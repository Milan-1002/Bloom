# Plan 03-02: Weight Logging

## Goal

A user can log today's weight (kg) from the Today dashboard, then see a 7-day rolling average sparkline on the weight card. Past entries are deduplicated by date (UPSERT). The rolling average — never raw daily values — is shown per CLAUDE.md rule 5.

## Context

- **Schema:** `weight_logs` — `user_id`, `log_date` (YYYY-MM-DD), `logged_at`, `weight_kg` (number). Unique: `(user_id, log_date)`.
- **Sparkline component:** `src/components/ui/Sparkline.tsx` — accepts `data: number[]`, `w`, `h`, `color`, `fill`, `dots`.
- **Pattern:** Same UPSERT-by-date pattern as symptom_logs (03-01).
- **CLAUDE.md rule 5:** Weight chart uses 7-day rolling average — never raw daily values.
- **No dedicated tab:** Weight entry is a bottom sheet on the HomeScreen weight card. Tab count stays at 4 (Home | Diary | Symptoms | Profile).

## Threat Model

| ID | Threat | Mitigation |
|----|--------|------------|
| T-03-02-01 | UPSERT another user's weight_log | RLS enforces `(select auth.uid()) = user_id`; `user_id` hardcoded from `user!.id` |
| T-03-02-02 | Negative or biologically impossible weight | Sheet input: `min="20"`, `max="300"`, `step="0.1"`; Zod validates before submit |

#### 1a — `rollingAverage` (pure function, no mocks needed)

Exported from `src/lib/rolling-average.ts`.

```ts
// rollingAverage(values, window): for each index i, average the last `window`
// elements up to and including i (or all available if fewer than window).
export function rollingAverage(values: number[], window: number): number[] {
  return values.map((_, i) => {
    const slice = values.slice(Math.max(0, i - window + 1), i + 1)
    return Math.round((slice.reduce((a, b) => a + b, 0) / slice.length) * 10) / 10
  })
}
```

Tests:
```ts
describe('rollingAverage', () => {
  it('returns the single value for a 1-element array', () => {
    expect(rollingAverage([65.0], 7)).toEqual([65.0])
  })

  it('averages all available when window is larger than array', () => {
    // [60, 62, 64] window=7 → all 3 averaged each time
    expect(rollingAverage([60, 62, 64], 7)).toEqual([60, 61, 62])
  })

  it('applies window correctly for a 7-day window over 10 values', () => {
    const data = [64, 64, 63, 63, 62, 62, 61, 61, 60, 60]
    const result = rollingAverage(data, 7)
    // index 6: avg of [64,64,63,63,62,62,61] = 62.7
    expect(result[6]).toBe(62.7)
    // index 9 (last): avg of [63,62,62,61,61,60,60] = 61.3
    expect(result[9]).toBe(61.3)
  })
})
```


Implementation:
```ts
export function useWeightLogs() {
  return useQuery({
    queryKey: ['weight-logs'],
    queryFn: async () => {
      const { data, error } = await supabase
        .from('weight_logs')
        .select('id, log_date, weight_kg, logged_at')
        .order('log_date', { ascending: true })
        .limit(30)
      if (error) throw error
      return (data ?? []) as Pick<Tables<'weight_logs'>, 'id' | 'log_date' | 'weight_kg' | 'logged_at'>[]
    },
    staleTime: 60 * 1000,
  })
}
```

#### 1c — `useLogWeight`

Supabase chain: `from → upsert`

```ts
// Test:
it('calls upsert with user_id, log_date, weight_kg and onConflict: user_id,log_date')
```

Implementation:
```ts
export interface LogWeightPayload {
  log_date: string   // YYYY-MM-DD
  weight_kg: number
}

export function useLogWeight() {
  const { user } = useAuth()
  const queryClient = useQueryClient()
  return useMutation({
    mutationFn: async ({ log_date, weight_kg }: LogWeightPayload) => {
      const { error } = await supabase.from('weight_logs').upsert(
        { user_id: user!.id, log_date, weight_kg, logged_at: new Date().toISOString() },
        { onConflict: 'user_id,log_date' },
      )
      if (error) throw error
    },
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: ['weight-logs'] })
    },
  })
}
```

### Task 3 — HomeScreen weight card + wire-up

**Modify:** `src/screens/home/HomeScreen.tsx`

Add `WeightCard` (file-local) after the `SymptomSummaryCard`:

```tsx
function WeightCard() {
  const [showEntry, setShowEntry] = useState(false)
  const today = toLocalDateStr()
  const { data: logs = [] } = useWeightLogs()

  const weights = logs.map(l => l.weight_kg)
  const avgSeries = rollingAverage(weights, 7)
  const currentAvg = avgSeries.at(-1) ?? null
  const prevAvg = avgSeries.length >= 8 ? avgSeries[avgSeries.length - 8] : null
  const delta = currentAvg != null && prevAvg != null
    ? Math.round((currentAvg - prevAvg) * 10) / 10
    : null

  const todayLog = logs.find(l => l.log_date === today)

  return (
    <>
      <Card className="p-4">
        <div className="flex items-center gap-3">
          <div className="flex-1">
            <p className="text-[10px] font-bold uppercase tracking-[0.4px] text-b-ink-3">
              WEIGHT · 7-DAY AVG
            </p>
            {currentAvg != null ? (
              <div className="mt-1 flex items-baseline gap-2">
                <span className="text-[26px] font-bold tabular-nums leading-none text-b-ink">
                  {currentAvg}
                </span>
                <span className="text-[13px] text-b-ink-3">kg</span>
                {delta !== null && delta !== 0 && (
                  <Chip tone={delta < 0 ? 'mint' : 'neutral'} size="sm">
                    {delta > 0 ? '+' : ''}{delta} kg
                  </Chip>
                )}
              </div>
            ) : (
              <p className="mt-1 text-[14px] text-b-ink-3">No entries yet</p>
            )}
          </div>

          {/* Sparkline — rolling average series */}
          {avgSeries.length > 0 && (
            <Sparkline
              data={avgSeries}
              w={100}
              h={44}
              color="var(--b-mint)"
              fill
              dots
              dotColor="var(--b-mint)"
            />
          )}
        </div>

        <button
          onClick={() => setShowEntry(true)}
          className="mt-3 w-full rounded-b-pill border border-b-hairline bg-b-surface-2 py-2 text-[13px] font-bold text-b-ink active:opacity-70"
        >
          {todayLog ? `Today: ${todayLog.weight_kg} kg — Update` : 'Log today's weight'}
        </button>
      </Card>

      <WeightEntrySheet
        open={showEntry}
        onClose={() => setShowEntry(false)}
        todayDate={today}
        existingWeight={todayLog?.weight_kg}
      />
    </>
  )
}
```

Wire at the end of the HomeScreen scrollable body, after `SymptomSummaryCard`.
